version: '3.8'

services:
  # Banco de dados PostgreSQL
  database:
    build:
      context: ./DB
      dockerfile: Dockerfile
    container_name: igem-stock-db-dev
    environment:
      POSTGRES_DB: igem_stock_dev
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: postgres123
      POSTGRES_HOST_AUTH_METHOD: md5
    ports:
      - "5432:5432"
    volumes:
      - postgres_data_dev:/var/lib/postgresql/data
      - postgres_backups_dev:/var/lib/postgresql/backups
      - ./DB/init-scripts:/docker-entrypoint-initdb.d
    networks:
      - igem-network
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres -d igem_stock_dev"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 10s

  # Redis para cache e filas
  redis:
    build:
      context: ./CA
      dockerfile: Dockerfile
    container_name: igem-stock-redis-dev
    ports:
      - "6379:6379"
    environment:
      - REDIS_PASSWORD=${REDIS_PASSWORD:-}
      - REDIS_MAXMEMORY=${REDIS_MAXMEMORY:-256mb}
      - REDIS_MAXMEMORY_POLICY=${REDIS_MAXMEMORY_POLICY:-allkeys-lru}
    volumes:
      - redis_data_dev:/data
      - redis_logs_dev:/var/log/redis
      - redis_run_dev:/var/run/redis
      - ./CA/config/redis.conf:/etc/redis/redis.conf:ro
      - ./CA/scripts:/usr/local/bin/scripts:ro
    networks:
      - igem-network
    # Remove the command override to let entrypoint handle config
    # command: ["redis-server", "/etc/redis/redis.conf"]
    healthcheck:
      test: ["CMD", "/usr/local/bin/healthcheck.sh"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 10s

  # Backend NestJS
  backend:
    build:
      context: ./BE-IGEMSTOCK
      dockerfile: Dockerfile.dev
    container_name: igem-stock-backend-dev
    environment:
      NODE_ENV: development
      DATABASE_URL: postgresql://postgres:postgres123@database:5432/igem_stock_dev?schema=public
      REDIS_URL: redis://redis:6379
      JWT_SECRET: dev_jwt_secret_key_change_in_production
      PORT: 3000
    ports:
      - "3000:3000"
    volumes:
      - ./BE-IGEMSTOCK:/app
      - /app/node_modules
    networks:
      - igem-network
    depends_on:
      database:
        condition: service_healthy
      redis:
        condition: service_healthy
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:3000/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 40s
    restart: unless-stopped

  # pgAdmin - Interface web para PostgreSQL
  pgadmin:
    image: dpage/pgadmin4:latest
    container_name: igem-pgadmin-dev
    environment:
      PGADMIN_DEFAULT_EMAIL: admin@igemstock.com
      PGADMIN_DEFAULT_PASSWORD: admin123
      PGADMIN_CONFIG_SERVER_MODE: 'False'
      PGADMIN_CONFIG_MASTER_PASSWORD_REQUIRED: 'False'
    ports:
      - "5050:80"
    volumes:
      - pgadmin_data_dev:/var/lib/pgadmin
      - ./DB/pgadmin/servers.json:/pgadmin4/servers.json:ro
    networks:
      - igem-network
    depends_on:
      database:
        condition: service_healthy
    restart: unless-stopped

  # RedisInsight - Interface web para Redis
  redisinsight:
    image: redislabs/redisinsight:latest
    container_name: igem-redisinsight-dev
    ports:
      - "8001:8001"
    volumes:
      - redisinsight_data_dev:/db
    environment:
      - RIPORT=8001
      - RITRUSTEDORIGINS=http://localhost:8001

      - igem-network
    depends_on:
      redis:
        condition: service_healthy
    restart: unless-stopped

  # Adminer - Interface web alternativa para banco de dados (leve)

      database:
